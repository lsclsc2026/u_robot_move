cmake_minimum_required(VERSION 3.8)
project(u_robot_teleop)
find_package(ament_cmake REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(TELEOP_UPSTREAM_SOURCE "" CACHE PATH "Full original a2_joystick_lab source tree, including CMakeLists.txt")
set(TELEOP_NATIVE_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../sdk/unitree_sdk2/thirdparty/lib/x86_64"
    CACHE PATH "Native receiver CycloneDDS libraries")
if(TELEOP_UPSTREAM_SOURCE)
  if(NOT EXISTS "${TELEOP_UPSTREAM_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Original source project is missing CMakeLists.txt")
  endif()
  add_subdirectory("${TELEOP_UPSTREAM_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/upstream")
  if(NOT TARGET a2_network_bridge)
    message(FATAL_ERROR "Original project must provide a2_network_bridge target")
  endif()
  install(TARGETS a2_network_bridge RUNTIME DESTINATION lib/${PROJECT_NAME})
else()
  message(WARNING "Installing preserved receiver binary; original PC1 C++ source has not been imported/rebuilt")
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/bin/a2_network_bridge")
    message(FATAL_ERROR "Missing receiver. Run scripts/deploy_teleop_docker.sh to stage the deployed binary")
  endif()
  install(PROGRAMS vendor/bin/a2_network_bridge DESTINATION lib/${PROJECT_NAME})
endif()
foreach(native_lib libddsc.so.0 libddscxx.so.0)
  if(NOT EXISTS "${TELEOP_NATIVE_LIB_DIR}/${native_lib}")
    message(FATAL_ERROR "Missing native SDK dependency: ${TELEOP_NATIVE_LIB_DIR}/${native_lib}")
  endif()
  get_filename_component(native_real "${TELEOP_NATIVE_LIB_DIR}/${native_lib}" REALPATH)
  install(FILES "${native_real}" DESTINATION lib/${PROJECT_NAME}/native RENAME "${native_lib}")
endforeach()
install(PROGRAMS scripts/teleop_receiver.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/teleop_runtime.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES vendor/PROVENANCE.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_test(NAME teleop_runtime COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/test -v)
  set_tests_properties(teleop_runtime PROPERTIES TIMEOUT 60)
endif()
ament_package()
